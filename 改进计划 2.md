## 一、优化目标与原则

### 1.1 核心目标

1. **修复 CloudKit 同步问题** - 确保数据可靠同步
2. **完善核心功能** - 数据导出、生物识别、错误处理
3. **优化架构** - 统一数据流，提高可维护性
4. **清理冗余代码** - 保持代码整洁

### 1.2 设计原则

| 原则 | 描述 | 实施标准 |
|------|------|---------|
| 单一职责 | 一个模块一个功能 | 每个文件只做一件事 |
| 高内聚低耦合 | 模块内聚，模块解耦 | 通过协议通信，避免直接依赖 |
| 命名清晰 | 语义化命名 | 见名知意，遵循 Swift 命名规范 |
| 代码复用 | 优先复用现有代码 | 三次法则：遇到三次再抽象 |
| 最小代码 | 不写多余代码 | 简洁优于复杂 |
| 风格一致 | 与现有代码风格统一 | SwiftUI 优先，遵循 Apple HIG |

import Foundation

/// Data export service
@MainActor
final class ExportService {
    static let shared = ExportService()
    
    private let modelContext: ModelContext
    
    init(modelContext: ModelContext) {
        self.modelContext = modelContext
    }
    
    /// Export transactions to file
    func exportTransactions(
        format: ExportFormat,
        startDate: Date? = nil,
        endDate: Date? = nil,
        completion: @escaping (Result<URL, Error>) -> Void
    ) {
        Task {
            do {
                var transactions = fetchTransactions(startDate: startDate, endDate: endDate)
                
                let fileURL: URL
                switch format {
                case .csv:
                    fileURL = try generateCSV(transactions: transactions)
                case .json:
                    fileURL = try generateJSON(transactions: transactions)
                case .pdf:
                    fileURL = try await generatePDF(transactions: transactions)
                }
                
                completion(.success(fileURL))
            } catch {
                completion(.failure(error))
            }
        }
    }
    
    // MARK: - Private Methods
    
    private func fetchTransactions(startDate: Date?, endDate: Date?) -> [Transaction] {
        let descriptor = FetchDescriptor<Transaction>(
            sortBy: [SortDescriptor(\.date, order: .reverse)]
        )
        
        var transactions = (try? modelContext.fetch(descriptor)) ?? []
        
        if let start = startDate {
            transactions = transactions.filter { $0.date >= start }
        }
        
        if let end = endDate {
            transactions = transactions.filter { $0.date <= end }
        }
        
        return transactions
    }
    
    private func generateCSV(transactions: [Transaction]) throws -> URL {
        var csvString = "Date,Amount,Note,Category,Type\n"
        
        for transaction in transactions {
            let date = DateFormatter.localizedString(from: transaction.date, dateStyle: .short, timeStyle: .none)
            let amount = String(format: "%.2f", transaction.amount)
            let note = transaction.note.replacingOccurrences(of: ",", with: "")
            let category = transaction.category?.name ?? "Uncategorized"
            let type = transaction.amount >= 0 ? "Income" : "Expense"
            
            csvString += "\(date),\(amount),\(note),\(category),\(type)\n"
        }
        
        return try saveToFile(content: csvString, extension: "csv")
    }
    
    private func generateJSON(transactions: [Transaction]) throws -> URL {
        let encoder = JSONEncoder()
        encoder.dateEncodingStrategy = .iso8601
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        
        let exportData = transactions.map { transaction in
            return ExportTransactionData(
                id: transaction.id,
                date: transaction.date,
                amount: transaction.amount,
                note: transaction.note,
                category: transaction.category?.name,
                isIncome: transaction.amount >= 0
            )
        }
        
        let jsonData = try encoder.encode(exportData)
        let jsonString = String(data: jsonData, encoding: .utf8) ?? ""
        
        return try saveToFile(content: jsonString, extension: "json")
    }
    
    private func generatePDF(transactions: [Transaction]) async throws -> URL {
        // PDF generation implementation
        // Using UIGraphicsImageRenderer or third-party library
        let pdfURL = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("PennyWise_Report_\(Date().ISO8601Format()).pdf")
        
        // TODO: Implement PDF generation
        return pdfURL
    }
    
    private func saveToFile(content: String, `extension`: String) throws -> URL {
        let filename = "PennyWise_Export_\(Date().ISO8601Format()).\(`extension`)"
        let fileURL = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
            .appendingPathComponent(filename)
        
        try content.write(to: fileURL, atomically: true, encoding: .utf8)
        return fileURL
    }
}

struct ExportDataView: View {

    @State private var dateRange: DateRange = .all

    @State private var errorMessage = ""

                                    .font(.headline)
                                Text(format.description)
                                    .font(.caption)
                                    .foregroundStyle(.secondary)
                            }
                            .tag(format)
                        }
                    }
                    .pickerStyle(.radioGroup)
                }
                
                Section("Date Range") {
                    Picker("Date Range", selection: $dateRange) {
                        Text("All Time").tag(DateRange.all)
                        Text("Last 30 Days").tag(DateRange.last30Days)
                        Text("Last 90 Days").tag(DateRange.last90Days)
                        Text("This Year").tag(DateRange.thisYear)
                        Text("Custom").tag(DateRange.custom)
                    }
                    
                    if dateRange == .custom {
                        DatePicker("From", selection: $startDate, displayedComponents: .date)
                        DatePicker("To", selection: $endDate, displayedComponents: .date)
                    }
                }
                
                Section {
                    Button {
                        exportData()
                    } label: {
                        HStack {
                            Spacer()
                            if isExporting {
                                ProgressView()
                                    .progressViewStyle(CircularProgressViewStyle())
                                Text("Exporting...")
                            } else {
                                Text("Export Data")
                                    .fontWeight(.semibold)
                            }
                            Spacer()
                        }
                    }
                    .disabled(isExporting)
                }
                
                if exportSuccess {
                    Section {
                        Button {
                            shareFile()
                        } label: {
                            Label("Share File", systemImage: "square.and.arrow.up")
                                .frame(maxWidth: .infinity)
                        }
                    }
                }
            }
            .navigationTitle("Export Data")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .confirmationAction) {
                    Button("Done") {
                        dismiss()
                    }
                }
            }
            .alert("Export Failed", isPresented: $showError) {
                Button("OK", role: .cancel) { }
            } message: {
                Text(errorMessage)
            }
        }
    }
    
    private func exportData() {
        isExporting = true
        
        let exportService = ExportService(modelContext: modelContext)
        let (startDate, endDate) = getDateRange()
        
        exportService.exportTransactions(
            format: selectedFormat,
            startDate: startDate,
            endDate: endDate
        ) { result in
            Task { @MainActor in
                isExporting = false
                
                switch result {
                case .success(let url):
                    exportedFileURL = url
                    exportSuccess = true
                case .failure(let error):
                    errorMessage = error.localizedDescription
                    showError = true
                }
            }
        }
    }
    
    private func shareFile() {
        guard let url = exportedFileURL else { return }
        
        let activityVC = UIActivityViewController(
            activityItems: [url],
            applicationActivities: nil
        )
        
        if let windowScene = UIApplication.shared.connectedScenes.first as? UIWindowScene,
           let rootViewController = windowScene.windows.first?.rootViewController {
            rootViewController.present(activityVC, animated: true)
        }
    }
    
    private func getDateRange() -> (Date?, Date?) {
        let calendar = Calendar.current
        let now = Date()
        
        switch dateRange {
        case .all:
            return (nil, nil)
        case .last30Days:
            return (calendar.date(byAdding: .day, value: -30, to: now), now)
        case .last90Days:
            return (calendar.date(byAdding: .day, value: -90, to: now), now)
        case .thisYear:
            return (calendar.date(from: calendar.dateComponents([.year], from: now)), now)
        case .custom:
            return (startDate, endDate)
        }
    }
}

    case last30Days = "Last 30 Days"
    case last90Days = "Last 90 Days"
    case thisYear = "This Year"
    case custom = "Custom"
    
    var id: String { rawValue }
}

import SwiftUI

struct ToastView: View {
    let message: String
    let type: ToastType
    let onDismiss: () -> Void
    
    @State private var offset: CGFloat = 100
    @State private var opacity: Double = 0
    
    var body: some View {
        HStack {
            Image(systemName: type.iconName)
            Text(message)
        }
        .font(.subheadline)
        .fontWeight(.medium)
        .foregroundColor(type.textColor)
        .padding(.horizontal, 20)
        .padding(.vertical, 12)
        .background(type.backgroundColor)
        .cornerRadius(12)
        .shadow(color: .black.opacity(0.15), radius: 8, x: 0, y: 4)
        .offset(y: offset)
        .opacity(opacity)
        .onAppear {
            withAnimation(.spring(response: 0.4, dampingFraction: 0.7)) {
                offset = 0
                opacity = 1
            }
            
            DispatchQueue.main.asyncAfter(deadline: .now() + 2.5) {
                dismiss()
            }
        }
    }
    
    private func dismiss() {
        withAnimation(.easeOut(duration: 0.3)) {
            offset = 100
            opacity = 0
        } completion: {
            onDismiss()
        }
    }
}

// MARK: - Toast Type

enum ToastType {
    case success
    case error
    case warning
    case info
    
    var iconName: String {
        switch self {
        case .success: return "checkmark.circle.fill"
        case .error: return "exclamationmark.circle.fill"
        case .warning: return "exclamationmark.triangle.fill"
        case .info: return "info.circle.fill"
        }
    }
    
    var textColor: Color {
        switch self {
        case .success: return .white
        case .error: return .white
        case .warning: return .black
        case .info: return .white
        }
    }
    
    var backgroundColor: Color {
        switch self {
        case .success: return Color(hex: "#34C759")
        case .error: return Color(hex: "#FF3B30")
        case .warning: return Color(hex: "#FF9500")
        case .info: return Color(hex: "#007AFF")
        }
    }
}

// MARK: - View Extension

extension View {
    func toast(message: Binding<String?>, type: ToastType = .info) -> some View {
        modifier(ToastModifier(message: message, type: type))
    }
}

struct ToastModifier: ViewModifier {
    @Binding var message: String?
    let type: ToastType
    
    func body(content: Content) -> some View {
        ZStack {
            content
            
            if let message = message {
                ToastView(
                    message: message,
                    type: type,
                    onDismiss: {
                        self.message = nil
                    }
                )
            }
        }
    }
}

import Foundation
import CloudKit
import SwiftData

/// Application error types
enum AppError: LocalizedError {
    case cloudKitUnavailable
    case cloudKitSyncFailed(Error)
    case dataSaveFailed(Error)
    case dataLoadFailed(Error)
    case exportFailed(Error)
    case importFailed(Error)
    case biometricAuthFailed
    case networkError
    case unknown(Error)
    
    var errorDescription: String? {
        switch self {
        case .cloudKitUnavailable:
            return "iCloud is not available. Please check your iCloud settings."
        case .cloudKitSyncFailed:
            return "Failed to sync with iCloud. Please try again later."
        case .dataSaveFailed:
            return "Failed to save data. Please try again."
        case .dataLoadFailed:
            return "Failed to load data. Please try again."
        case .exportFailed(let error):
            return "Export failed: \(error.localizedDescription)"
        case .importFailed:
            return "Import failed. Please check the file format."
        case .biometricAuthFailed:
            return "Authentication failed. Please try again."
        case .networkError:
            return "Network error. Please check your connection."
        case .unknown(let error):
            return "An unexpected error occurred: \(error.localizedDescription)"
        }
    }
    
    var recoverySuggestion: String? {
        switch self {
        case .cloudKitUnavailable:
            return "Make sure you're signed in to iCloud and have an internet connection."
        case .cloudKitSyncFailed:
            return "Your data is saved locally and will sync when iCloud is available."
        case .dataSaveFailed:
            return "Make sure you have enough storage space."
        default:
            return nil
        }
    }
    
    init(_ error: Error) {
        if let ckError = error as? CKError {
            switch ckError.code {
            case .notAuthenticated:
                self = .cloudKitUnavailable
            case .networkUnavailable:
                self = .networkError
            case .quotaExceeded:
                self = .cloudKitSyncFailed(error)
            default:
                self = .cloudKitSyncFailed(error)
            }
        } else if let swiftDataError = error as? SwiftDataError {
            switch swiftDataError {
            case .loadIssueModelContainer:
                self = .dataLoadFailed(error)
            default:
                self = .dataSaveFailed(error)
            }
        } else {
            self = .unknown(error)
        }
    }
}

/// Error handler singleton
final class ErrorHandler {
    static let shared = ErrorHandler()
    
    /// Handle error and show user feedback
    func handle(_ error: Error, context: String = "") {
        let appError = AppError(error)
        
        // Log error
        print("Error in \(context): \(appError)")
        if let recovery = appError.recoverySuggestion {
            print("Recovery suggestion: \(recovery)")
        }
        
        // Post notification for UI to show
        NotificationCenter.default.post(
            name: .errorOccurred,
            object: nil,
            userInfo: [
                "error": appError,
                "context": context
            ]
        )
    }
    
    private init() {}
}

// MARK: - Notification Name

extension Notification.Name {
    static let errorOccurred = Notification.Name("errorOccurred")
}

import Foundation

/// Biometric authentication service
@MainActor
final class BiometricService {
    static let shared = BiometricService()
    
    private let context = LAContext()
    private let userDefaults = UserDefaults.standard
    
    /// Check if biometric authentication is available
    var isBiometricAvailable: Bool {
        var error: NSError?
        return context.canEvaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, error: &error)
    }
    
    /// Check if biometric lock is enabled
    var isBiometricLockEnabled: Bool {
        userDefaults.bool(forKey: "biometricLockEnabled")
    }
    
    /// Enable/disable biometric lock
    func setBiometricLockEnabled(_ enabled: Bool) {
        userDefaults.set(enabled, forKey: "biometricLockEnabled")
    }
    
    /// Authenticate with biometrics
    func authenticate(reason: String = "Authenticate to access PennyWise") async -> Bool {
        var error: NSError?
        
        guard context.canEvaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, error: &error) else {
            print("Biometric authentication not available: \(error?.localizedDescription ?? "Unknown error")")
            return false
        }
        
        do {
            let success = try await context.evaluatePolicy(
                .deviceOwnerAuthenticationWithBiometrics,
                localizedReason: reason
            )
            return success
        } catch {
            print("Authentication failed: \(error)")
            return false
        }
    }
    
    /// Authenticate with fallback to passcode
    func authenticateWithFallback(reason: String = "Authenticate to access PennyWise") async -> Bool {
        var error: NSError?
        
        guard context.canEvaluatePolicy(.deviceOwnerAuthentication, error: &error) else {
            print("Authentication not available: \(error?.localizedDescription ?? "Unknown error")")
            return false
        }
        
        do {
            let success = try await context.evaluatePolicy(
                .deviceOwnerAuthentication,
                localizedReason: reason
            )
            return success
        } catch {
            print("Authentication failed: \(error)")
            return false
        }
    }
    
    private init() {}
}

        }
        .onAppear {
            loadBiometricType()
        }
    }
    
    private var biometricTypeName: String {
        switch biometricType {
        case .faceID: return "Face ID"
        case .touchID: return "Touch ID"
        case .none, @unknown default: return "Biometric"
        }
    }
    
    private var biometricTypeIcon: String {
        switch biometricType {
        case .faceID: return "faceid"
        case .touchID: return "touchid"
        case .none, @unknown default: return "lock.shield"
        }
    }
    
    private func loadBiometricType() {
        let context = LAContext()
        var error: NSError?
        
        if context.canEvaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, error: &error) {
            biometricType = context.biometryType
        } else {
            biometricType = .none
        }
    }

// MARK: - Biometric Setup View

struct BiometricSetupView: View {
    @Environment(\.dismiss) private var dismiss
    @Binding var isEnabled: Bool
    
    @State private var isAuthenticating = false
    @State private var authSuccess = false
    @State private var authError: String?
    
    var body: some View {
        NavigationStack {
            VStack(spacing: 30) {
                Spacer()
                
                Image(systemName: "lock.shield")
                    .font(.system(size: 80))
                    .foregroundStyle(.primary)
                
                Text("Enable App Lock")
                    .font(.title)
                    .fontWeight(.bold)
                
                Text("You'll need to use Face ID to open the app and protect your data")
                    .font(.body)
                    .foregroundStyle(.secondary)
                    .multilineTextAlignment(.center)
                
                if let error = authError {
                    Text(error)
                        .font(.caption)
                        .foregroundStyle(.red)
                }
                
                Button {
                    authenticate()
                } label: {
                    Text(isAuthenticating ? "Authenticating..." : "Enable Face ID")
                        .fontWeight(.semibold)
                        .frame(maxWidth: .infinity)
                        .padding()
                        .background(authSuccess ? Color.green : Color.blue)
                        .foregroundStyle(.white)
                        .cornerRadius(12)
                }
                .disabled(isAuthenticating || !authSuccess)
                
                Spacer()
            }
            .padding()
            .navigationTitle("Setup App Lock")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .confirmationAction) {
                    Button("Done") {
                        dismiss()
                    }
                }
            }
        }
    }
    
    private func authenticate() {
        isAuthenticating = true
        
        Task {
            let success = await BiometricService.shared.authenticate()
            
            await MainActor.run {
                isAuthenticating = false
                authSuccess = success
                
                if success {
                    BiometricService.shared.setBiometricLockEnabled(true)
                    DispatchQueue.main.asyncAfter(deadline: .now() + 1) {
                        dismiss()
                    }
                } else {
                    authError = "Authentication failed. Please try again."
                }
            }
        }
    }
}
