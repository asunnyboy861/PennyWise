## 一、项目概述与市场分析

### 1.1 项目背景

简约记账追踪器是一款面向美国市场的轻量级个人财务管理应用。该项目基于对现有记账应用市场的深入分析，针对用户痛点进行差异化设计。当前美国记账应用市场存在以下问题：

1. **订阅疲劳**: 大多数主流记账应用采用订阅制（如Mint、YNAB、Personal Capital），用户每月需支付$5-$15不等
2. **过度复杂**: 功能堆砌严重，入门门槛高，新用户流失率高
3. **隐私担忧**: 银行账户连接带来数据安全隐患
4. **数据锁定**: 用户难以导出自己的数据，存在供应商锁定问题
5. **广告干扰**: 免费应用充斥广告，影响用户体验

### 1.2 核心价值主张

| 价值主张 | 描述 | 差异化优势 |
|---------|------|-----------|
| 离线优先 | 数据完全本地存储，无需联网 | 100%隐私保护 |
| 一次性付费 | 只需$3.99，无订阅负担 | 永久使用，无后续费用 |
| 3秒记账 | 极简操作流程，快速记录支出 | 比竞品快50% |

```swift
// Transaction.swift - 交易记录模型
import Foundation
import CoreData

@objc(Transaction)
public class Transaction: NSManagedObject {
    @NSManaged public var id: UUID
    @NSManaged public var amount: Double
    @NSManaged public var note: String
    @NSManaged public var date: Date
    @NSManaged public var category: ExpenseCategory
    @NSManaged public var isRecurring: Bool       // 是否重复
    @NSManaged public var recurringInterval: String? // 重复间隔
    @NSManaged public var attachmentData: Data?   // 附件图片
    @NSManaged public var location: String?     // 位置信息
    
    // 计算属性
    var formattedAmount: String {
        let formatter = NumberFormatter()
        formatter.numberStyle = .currency
        formatter.currencyCode = "USD"
        return formatter.string(from: NSNumber(value: amount)) ?? "$0.00"
    }
    
    var formattedDate: String {
        let formatter = DateFormatter()
        formatter.dateStyle = .medium
        formatter.timeStyle = .short
        return formatter.string(from: date)
    }
}
```

```swift
    var spentAmount: Double {
        // 计算当前已花费
        return 0.0 // 在ViewModel中实现
    }
    
    var remainingAmount: Double {
        return limitAmount - spentAmount
    }

    }
```

```swift
import SwiftUI

struct AddTransactionView: View {
    @ObservedObject var viewModel: DashboardViewModel
    @Environment(\.dismiss) private var dismiss
    
    @State private var amount: String = ""
    @State private var selectedCategory: ExpenseCategory?
    @State private var note: String = ""
    @State private var date: Date = Date()
    @State private var isExpense: Bool = true
    
    // 快速金额按钮
    private let quickAmounts: [Double] = [5, 10, 20, 50, 100]
    
    var body: some View {
        NavigationStack {
            Form {
                // 金额输入区
                Section("Amount") {
                    HStack {
                        Text("$")
                            .font(.title)
                            .foregroundColor(.secondary)
                        
                        TextField("0.00", text: $amount)
                            .font(.system(size: 40, weight: .bold))
                            .keyboardType(.decimalPad)
                    }
                    .padding(.vertical, 8)
                    
                    // 快速金额按钮
                    LazyVGrid(columns: [
                        GridItem(.flexible()),
                        GridItem(.flexible()),
                        GridItem(.flexible()),
                        GridItem(.flexible()),
                        GridItem(.flexible())
                    ], spacing: 8) {
                        ForEach(quickAmounts, id: \.self) { quickAmount in
                            Button {
                                amount = String(format: "%.2f", quickAmount)
                            } label: {
                                Text("$\(Int(quickAmount))")
                                    .font(.caption)
                                    .padding(.horizontal, 12)
                                    .padding(.vertical, 8)
                                    .background(Color.blue.opacity(0.1))
                                    .cornerRadius(8)
                            }
                            .buttonStyle(.plain)
                        }
                    }
                }
                
                // 收入/支出切换
                Section {
                    Picker("Type", selection: $isExpense) {
                        Text("Expense").tag(true)
                        Text("Income").tag(false)
                    }
                    .pickerStyle(.segmented)
                }
                
                // 类别选择
                Section("Category") {
                    CategoryGrid(
                        selectedCategory: $selectedCategory,
                        isExpense: isExpense
                    )
                }
                
                // 备注
                Section("Note (Optional)") {
                    TextField("Add a note...", text: $note)
                }
                
                // 日期
                Section("Date") {
                    DatePicker(
                        "Date",
                        selection: $date,
                        displayedComponents: [.date, .hourAndMinute]
                    )
                }
            }
            .navigationTitle("Add Transaction")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .navigationBarLeading) {
                    Button("Cancel") {
                        dismiss()
                    }
                }
                
                ToolbarItem(placement: .navigationBarTrailing) {
                    Button("Save") {
                        saveTransaction()
                    }
                    .disabled(amount.isEmpty || selectedCategory == nil)
                    .fontWeight(.semibold)
                }
            }
        }
    }
    
    private func saveTransaction() {
        guard let amountValue = Double(amount),
              let category = selectedCategory else { return }
        
        viewModel.addTransaction(
            amount: amountValue,
            category: category,
            note: note,
            date: date,
            isIncome: !isExpense
        )
        
        dismiss()
    }
}

// 类别选择网格
struct CategoryGrid: View {
    @Binding var selectedCategory: ExpenseCategory?
    let isExpense: Bool
    
    private let columns = [
        GridItem(.flexible()),
        GridItem(.flexible()),
        GridItem(.flexible()),
        GridItem(.flexible())
    ]
    
    var body: some View {
        LazyVGrid(columns: columns, spacing: 12) {
            ForEach(getCategories(), id: \.self) { category in
                CategoryButton(
                    category: category,
                    isSelected: selectedCategory?.id == category.id
                ) {
                    selectedCategory = category
                }
            }
        }
    }
    
    private func getCategories() -> [ExpenseCategory] {
        // 从ViewModel获取类别列表
        return ExpenseCategory.defaultCategories
            .filter { $0.isIncome != isExpense }
            .map { cat in
                let category = ExpenseCategory()
                category.name = cat.name
                category.icon = cat.icon
                category.colorHex = cat.color
                category.isIncome = cat.isIncome
                return category
            }
    }
}

extension Color {

        let a, r, g, b: UInt64
        switch hex.count {
        case 3:
            (a, r, g, b) = (255, (int >> 8) * 17, (int >> 4 & 0xF) * 17, (int & 0xF) * 17)
        case 6:
            (a, r, g, b) = (255, int >> 16, int >> 8 & 0xFF, int & 0xFF)
        case 8:
            (a, r, g, b) = (int >> 24, int >> 16 & 0xFF, int >> 8 & 0xFF, int & 0xFF)
        default:
            (a, r, g, b) = (255, 0, 0, 0)
        }
        self.init(
            .sRGB,
            red: Double(r) / 255,
            green: Double(g) / 255,
            blue: Double(b) / 255,
            opacity: Double(a) / 255
        )
    }
```

```swift
import SwiftUI
import CoreData
import Combine

class DashboardViewModel: ObservableObject {
    // MARK: - Published Properties
    @Published var todayTotal: Double = 0
    @Published var todayIncome: Double = 0
    @Published var todayExpense: Double = 0
    @Published var weeklyData: [DailyExpense] = []
    @Published var recentTransactions: [Transaction] = []
    @Published var categories: [ExpenseCategory] = []
    @Published var isLoading: Bool = false
    @Published var errorMessage: String?
    
    // MARK: - Private Properties
    private let viewContext: NSManagedObjectContext
    private var cancellables = Set<AnyCancellable>()
    
    // MARK: - Initialization
    init(context: NSManagedObjectContext = PersistenceController.shared.viewContext) {
        self.viewContext = context
        loadData()
    }
    
    // MARK: - Public Methods
    func loadData() {
        isLoading = true
        
        // 加载今日数据
        loadTodayData()
        
        // 加载本周数据
        loadWeeklyData()
        
        // 加载最近交易
        loadRecentTransactions()
        
        // 加载类别
        loadCategories()
        
        isLoading = false
    }
    
    func addTransaction(
        amount: Double,
        category: ExpenseCategory,
        note: String,
        date: Date,
        isIncome: Bool
    ) {
        let transaction = Transaction(context: viewContext)
        transaction.id = UUID()
        transaction.amount = amount
        transaction.category = category
        transaction.note = note
        transaction.date = date
        transaction.isRecurring = false
        
        // 根据类型设置正负值
        if !isIncome {
            transaction.amount = -abs(amount)
        }
        
        do {
            try viewContext.save()
            loadData()
        } catch {
            errorMessage = "Failed to save transaction: \(error.localizedDescription)"
        }
    }
    
    func deleteTransaction(_ transaction: Transaction) {
        viewContext.delete(transaction)
        
        do {
            try viewContext.save()
            loadData()
        } catch {
            errorMessage = "Failed to delete transaction: \(error.localizedDescription)"
        }
    }
    
    func exportData(to format: ExportFormat) -> URL? {
        // 实现数据导出逻辑
        let transactions = fetchAllTransactions()
        
        switch format {
        case .csv:
            return exportToCSV(transactions)
        case .json:
            return exportToJSON(transactions)
        case .pdf:
            return exportToPDF(transactions)
        }
    }
    
    // MARK: - Private Methods
    private func loadTodayData() {
        let calendar = Calendar.current
        let startOfDay = calendar.startOfDay(for: Date())
        let endOfDay = calendar.date(byAdding: .day, value: 1, to: startOfDay)!
        
        let request: NSFetchRequest<Transaction> = Transaction.fetchRequest()
        request.predicate = NSPredicate(
            format: "date >= %@ AND date < %@",
            startOfDay as NSDate,
            endOfDay as NSDate
        )
        
        do {
            let todayTransactions = try viewContext.fetch(request)
            
            todayIncome = todayTransactions
                .filter { $0.amount > 0 }
                .reduce(0) { $0 + $1.amount }
            
            todayExpense = abs(todayTransactions
                .filter { $0.amount < 0 }
                .reduce(0) { $0 + $1.amount })
            
            todayTotal = todayIncome - todayExpense
        } catch {
            errorMessage = "Failed to fetch today's data"
        }
    }
    
    private func loadWeeklyData() {
        let calendar = Calendar.current
        let today = Date()
        
        // 获取本周开始日期
        guard let weekStart = calendar.date(from: calendar.dateComponents([.yearForWeekOfYear, .weekOfYear], from: today)) else { return }
        
        var data: [DailyExpense] = []
        
        for dayOffset in 0..<7 {
            guard let dayDate = calendar.date(byAdding: .day, value: dayOffset, to: weekStart) else { continue }
            
            let request: NSFetchRequest<Transaction> = Transaction.fetchRequest()
            let startOfDay = calendar.startOfDay(for: dayDate)
            let endOfDay = calendar.date(byAdding: .day, value: 1, to: startOfDay)!
            
            request.predicate = NSPredicate(
                format: "date >= %@ AND date < %@",
                startOfDay as NSDate,
                endOfDay as NSDate
            )
            
            do {
                let dayTransactions = try viewContext.fetch(request)
                let total = dayTransactions.reduce(0) { $0 + $1.amount }
                
                let dayName = calendar.shortWeekdaySymbols[calendar.component(.weekday, from: dayDate) - 1]
                
                data.append(DailyExpense(
                    dayName: dayName,
                    date: dayDate,
                    amount: abs(total)
                ))
            } catch {
                data.append(DailyExpense(
                    dayName: calendar.shortWeekdaySymbols[dayOffset],
                    date: dayDate,
                    amount: 0
                ))
            }
        }
        
        weeklyData = data
    }
    
    private func loadRecentTransactions() {
        let request: NSFetchRequest<Transaction> = Transaction.fetchRequest()
        request.sortDescriptors = [NSSortDescriptor(keyPath: \Transaction.date, ascending: false)]
        request.fetchLimit = 10
        
        do {
            recentTransactions = try viewContext.fetch(request)
        } catch {
            errorMessage = "Failed to fetch recent transactions"
        }
    }
    
    private func loadCategories() {
        let request: NSFetchRequest<ExpenseCategory> = ExpenseCategory.fetchRequest()
        
        do {
            let existingCategories = try viewContext.fetch(request)
            
            if existingCategories.isEmpty {
                // 创建默认类别
                createDefaultCategories()
            } else {
                categories = existingCategories
            }
        } catch {
            errorMessage = "Failed to fetch categories"
        }
    }
    
    private func createDefaultCategories() {
        for defaultCat in ExpenseCategory.defaultCategories {
            let category = ExpenseCategory(context: viewContext)
            category.id = UUID()
            category.name = defaultCat.name
            category.icon = defaultCat.icon
            category.colorHex = defaultCat.color
            category.isIncome = defaultCat.isIncome
        }
        
        do {
            try viewContext.save()
            loadCategories()
        } catch {
            errorMessage = "Failed to create default categories"
        }
    }
    
    private func fetchAllTransactions() -> [Transaction] {
        let request: NSFetchRequest<Transaction> = Transaction.fetchRequest()
        request.sortDescriptors = [NSSortDescriptor(keyPath: \Transaction.date, ascending: false)]
        
        do {
            return try viewContext.fetch(request)
        } catch {
            return []
        }
    }
    
    // MARK: - Export Methods
    private func exportToCSV(_ transactions: [Transaction]) -> URL? {
        var csvString = "Date,Amount,Category,Note,Type\n"
        
        for transaction in transactions {
            let type = transaction.amount >= 0 ? "Income" : "Expense"
            let dateFormatter = DateFormatter()
            dateFormatter.dateFormat = "yyyy-MM-dd HH:mm"
            
            csvString += "\"\(dateFormatter.string(from: transaction.date))\","
            csvString += "\(abs(transaction.amount)),"
            csvString += "\"\(transaction.category.name)\","
            csvString += "\"\(transaction.note)\","
            csvString += "\(type)\n"
        }
        
        // 保存到临时文件
        let fileName = "expenses_export_\(Date().timeIntervalSince1970).csv"
        let fileURL = FileManager.default.temporaryDirectory.appendingPathComponent(fileName)
        
        do {
            try csvString.write(to: fileURL, atomically: true, encoding: .utf8)
            return fileURL
        } catch {
            return nil
        }
    }
    
    private func exportToJSON(_ transactions: [Transaction]) -> URL? {
        var jsonArray: [[String: Any]] = []
        
        let dateFormatter = DateFormatter()
        dateFormatter.dateFormat = "yyyy-MM-dd HH:mm"
        
        for transaction in transactions {
            jsonArray.append([
                "id": transaction.id.uuidString,
                "date": dateFormatter.string(from: transaction.date),
                "amount": abs(transaction.amount),
                "type": transaction.amount >= 0 ? "income" : "expense",
                "category": transaction.category.name,
                "note": transaction.note
            ])
        }
        
        do {
            let jsonData = try JSONSerialization.data(withJSONObject: jsonArray, options: .prettyPrinted)
            let fileName = "expenses_export_\(Date().timeIntervalSince1970).json"
            let fileURL = FileManager.default.temporaryDirectory.appendingPathComponent(fileName)
            try jsonData.write(to: fileURL)
            return fileURL
        } catch {
            return nil
        }
    }
    
    private func exportToPDF(_ transactions: [Transaction]) -> URL? {
        // PDF导出需要更复杂的实现，这里简化处理
        return nil
    }
}
```

### 7.1 设计原则

根据 Apple Human Interface Guidelines 和美国用户习惯，制定以下设计原则：

| 原则 | 描述 | 实现方式 |
|-----|------|---------|
| **清晰 (Clarity)** | 信息层次分明 | 大字体标题，高对比度 |
| **遵从 (Deference)** | 不干扰用户 | 简洁界面，无广告 |
| **深度 (Depth)** | 层级清晰 | 导航清晰，视觉层次 |

### 8.4 Widget 小组件实现

```swift
// Views/Widgets/ExpenseWidget.swift
import SwiftUI
import WidgetKit

struct ExpenseWidgetEntry: TimelineEntry {
    let date: Date
    let todayExpense: Double
    let weekExpense: Double
    let recentTransactions: [Transaction]
}

struct ExpenseWidgetProvider: TimelineProvider {
    func placeholder(in context: Context) -> ExpenseWidgetEntry {
        ExpenseWidgetEntry(
            date: Date(),
            todayExpense: 45.00,
            weekExpense: 230.00,
            recentTransactions: []
        )
    }
    
    func getSnapshot(in context: Context, completion: @escaping (ExpenseWidgetEntry) -> Void) {
        let entry = ExpenseWidgetEntry(
            date: Date(),
            todayExpense: 45.00,
            weekExpense: 230.00,
            recentTransactions: []
        )
        completion(entry)
    }
    
    func getTimeline(in context: Context, completion: @escaping (Timeline<ExpenseWidgetEntry>) -> Void) {
        // 从App Group读取数据
        let userDefaults = UserDefaults(suiteName: "group.com.minimalexpense.tracker")
        let todayExpense = userDefaults?.double(forKey: "todayExpense") ?? 0
        let weekExpense = userDefaults?.double(forKey: "weekExpense") ?? 0
        
        let entry = ExpenseWidgetEntry(
            date: Date(),
            todayExpense: todayExpense,
            weekExpense: weekExpense,
            recentTransactions: []
        )
        
        // 每小时更新一次
        let nextUpdate = Calendar.current.date(byAdding: .hour, value: 1, to: Date())!
        let timeline = Timeline(entries: [entry], policy: .after(nextUpdate))
        
        completion(timeline)
    }
}

struct ExpenseWidgetEntryView: View {
    var entry: ExpenseWidgetProvider.Entry
    @Environment(\.widgetFamily) var family

        switch family {
        case .systemSmall:

        case .systemMedium:
            MediumWidgetView(entry: entry)

            SmallWidgetView(entry: entry)
        }
    }
}
```
